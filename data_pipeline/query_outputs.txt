Q6 JOIN
======================================================================
SELECT
                c.category_name,
                b.title,
                b.star_rating,
                b.rating,
                b.price_gbp,
                b.price_inr
            FROM books b
            JOIN categories c
                ON b.category_id = c.category_id
            ORDER BY
                b.rating DESC,
                c.category_name,
                b.title
            LIMIT 10;
